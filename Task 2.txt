import numpy as np
import pandas as pd

np.random.seed(42)
sizes = [70, 80, 50]  # budget, regular, VIP customers

df = pd.DataFrame({
    "CustomerID": range(1, sum(sizes) + 1),
    "TotalSpend": np.concatenate([np.random.normal(m, s, n) for (m, s), n in zip([(200, 50), (800, 100), (2000, 250)], sizes)]),
    "PurchaseFrequency": np.concatenate([np.random.normal(m, s, n) for (m, s), n in zip([(3, 1), (12, 3), (30, 5)], sizes)]),
    "AvgBasketValue": np.concatenate([np.random.normal(m, s, n) for (m, s), n in zip([(40, 10), (70, 15), (120, 20)], sizes)]),
    "DaysSinceLastPurchase": np.concatenate([np.random.normal(m, s, n) for (m, s), n in zip([(90, 25), (35, 12), (7, 4)], sizes)]),
})

df["TotalSpend"] = df["TotalSpend"].clip(lower=20).round(2)
df["PurchaseFrequency"] = df["PurchaseFrequency"].clip(lower=1).round().astype(int)
df["AvgBasketValue"] = df["AvgBasketValue"].clip(lower=10).round(2)
df["DaysSinceLastPurchase"] = df["DaysSinceLastPurchase"].clip(lower=1).round().astype(int)

# shuffle so rows aren't ordered by segment
df = df.sample(frac=1, random_state=1).reset_index(drop=True)
df["CustomerID"] = range(1, len(df) + 1)
df.to_csv("customers.csv", index=False)
